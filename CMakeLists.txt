cmake_minimum_required(VERSION 3.24)
project(AmbianceGenerator VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE OFF CACHE BOOL "" FORCE)
function(archive name filename hash)
 FetchContent_Declare(${name} URL ${CMAKE_CURRENT_SOURCE_DIR}/vendor/${filename} URL_HASH SHA256=${hash} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endfunction()
archive(SDL SDL3-3.2.10.tar.gz f87be7b4dec66db4098e9c167b2aa34e2ca10aeb5443bdde95ae03185ed513e0)
archive(box2d box2d-3.1.1.tar.gz fb6ef914b50f4312d7d921a600eabc12318bb3c55a0b8c0b90608fa4488ef2e4)
archive(lua lua-5.4.8.tar.gz 4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
FetchContent_Declare(miniaudio URL ${CMAKE_CURRENT_SOURCE_DIR}/vendor/miniaudio-0.11.22.tar.gz URL_HASH SHA256=bcb07bfb27e6fa94d34da73ba2d5642d4940b208ec2a660dbf4e52e6b7cd492f DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR header-only)
FetchContent_MakeAvailable(SDL box2d lua miniaudio)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS ${lua_SOURCE_DIR}/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(lua54 STATIC ${LUA_SOURCES})
target_include_directories(lua54 PUBLIC ${lua_SOURCE_DIR}/src)
add_library(ambiance_core src/core.cpp src/audio.cpp)
target_include_directories(ambiance_core PUBLIC include)
target_link_libraries(ambiance_core PRIVATE box2d lua54)
add_library(ambiance_render src/render.cpp)
target_include_directories(ambiance_render PUBLIC include)
target_link_libraries(ambiance_render PUBLIC ambiance_core SDL3::SDL3)
option(AMBIANCE_CAPABILITY_ONLY "Build capability proof only" OFF)
if(AMBIANCE_CAPABILITY_ONLY)
 add_executable(capability tests/capability.cpp)
 target_link_libraries(capability PRIVATE ambiance_render)
 return()
endif()
add_library(ambiance_export src/export.cpp)
file(GLOB_RECURSE ENGINE_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ENGINE_FILES})
set(ENGINE_HASH_INPUT "")
foreach(source_file IN LISTS ENGINE_FILES)
 file(SHA256 ${source_file} source_hash)
 string(APPEND ENGINE_HASH_INPUT "${source_hash}")
endforeach()
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt cmake_hash)
string(SHA256 ENGINE_BUILD_ID "${ENGINE_HASH_INPUT}${cmake_hash}")
target_compile_definitions(ambiance_export PRIVATE AMBIANCE_BUILD_TYPE="${CMAKE_BUILD_TYPE}" AMBIANCE_BUILD_ID="${ENGINE_BUILD_ID}")
target_link_libraries(ambiance_export PUBLIC ambiance_render)
add_library(ambiance_preview src/preview.cpp src/device.cpp)
target_include_directories(ambiance_preview PRIVATE ${miniaudio_SOURCE_DIR})
target_link_libraries(ambiance_preview PUBLIC ambiance_render "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
add_executable(ambiance src/main.cpp)
target_link_libraries(ambiance PRIVATE ambiance_preview ambiance_export)
option(AMBIANCE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
foreach(t ambiance_core ambiance_render ambiance_preview ambiance_export ambiance)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
 if(AMBIANCE_SANITIZE)
  target_compile_options(${t} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ambiance_render)
add_test(NAME core_contracts COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME export_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:ambiance> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS ambiance RUNTIME DESTINATION bin)
add_executable(transport_tests tests/transport_tests.cpp)
target_link_libraries(transport_tests PRIVATE ambiance_preview)
add_test(NAME transport_contracts COMMAND transport_tests)
# Explicit desktop smoke check; not part of the hardware-free CTest suite.
add_executable(preview_smoke tests/preview_smoke.cpp)
target_link_libraries(preview_smoke PRIVATE ambiance_preview ambiance_export)
add_executable(pendulum_tests tests/pendulum_tests.cpp)
target_link_libraries(pendulum_tests PRIVATE ambiance_render)
add_test(NAME pendulum_contracts COMMAND pendulum_tests ${CMAKE_CURRENT_SOURCE_DIR})
